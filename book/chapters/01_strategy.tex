\chapter{ストラテジ系}

ストラテジ系は「経営・戦略・財務・法務」の分野。出題数は全100問中 約35問で\textbf{最も配点が大きい分野}。

% ====================================================
%  2.1 SWOT分析
% ====================================================
\section{経営戦略の分析手法}

\mondaipage
\unittheme{SWOT分析}{swot}

\begin{stumble}
「強み」と「機会」の区別がつかない人が多い。
\textbf{社内のこと＝強み・弱み}、\textbf{社外のこと＝機会・脅威}。
「内と外」の軸で分ける。
\end{stumble}

\begin{center}
\begin{tabularx}{\textwidth}{lXX}
  \toprule
  & \textbf{プラス面} & \textbf{マイナス面} \\
  \midrule
  \textbf{内部環境}（自社） & 強み (S) & 弱み (W) \\
  \textbf{外部環境}（市場） & 機会 (O) & 脅威 (T) \\
  \bottomrule
\end{tabularx}
\end{center}

\pitfallmemo{「新技術の登場」を強みにしがち → 自社が作ったのでなければ「機会」（外部のプラス）}

\begin{mondai}{SWOT分析で扱う4要素の組合せとして正しいものはどれか。}{basic}
  \choice{強み・弱み・機会・脅威}
  \choice{顧客・競合・自社・市場}
  \choice{製品・価格・流通・販促}
  \choice{理念・方針・戦略・戦術}
\end{mondai}

\begin{mondai}{ある企業がSWOT分析を行った。「競合他社が撤退して市場シェアが拡大する可能性がある」という要因は、SWOTのどれに分類されるか。}{standard}
  \choice{強み（Strengths）}
  \choice{弱み（Weaknesses）}
  \choice{機会（Opportunities）}
  \choice{脅威（Threats）}
\end{mondai}

\begin{mondai}{SWOT分析と3C分析の違いとして最も適切なものはどれか。}{trick}
  \choice{SWOT分析は外部環境のみ、3C分析は内部環境のみを分析する}
  \choice{SWOT分析は内部と外部の両面、3C分析は顧客・競合・自社の3視点で分析する}
  \choice{3C分析はSWOT分析の一部である}
  \choice{両者はまったく同じ分析手法の別名である}
\end{mondai}

\kaitoupage

\answer{A}
\begin{kaisetsu}
SWOTは \textbf{Strengths（強み）・Weaknesses（弱み）・Opportunities（機会）・Threats（脅威）}の頭文字。イは3C分析+α、ウは4P、エは経営階層の話。
\end{kaisetsu}

\answer{C}
\begin{kaisetsu}
「競合他社の撤退」は\textbf{外部環境の変化}であり、自社にとって\textbf{プラスの要因}。よって機会（O）に分類。
\end{kaisetsu}
\begin{goutoubun}
\begin{itemize}
  \item ア「強み」→ 内部のプラス。競合の動向ではない
  \item イ「弱み」→ 内部のマイナス。外部の話ではない
  \item ウ「機会」→ 外部のプラス。\textbf{正解}
  \item エ「脅威」→ 外部のマイナス。競合撤退はプラス
\end{itemize}
\end{goutoubun}

\answer{B}
\begin{kaisetsu}
SWOTは内部/外部の2軸。3Cは顧客・競合・自社の3視点。フレームワークの切り口が異なる。「3CはSWOTの一部」は誤り——独立したフレームワークで併用することが多い。
\end{kaisetsu}

\hitokoto{SWOT＝内と外×プラスとマイナス。「誰の話か」で分ける。}
\qrunit{2}{1}

% ====================================================
%  2.2 3C分析
% ====================================================
\mondaipage
\unittheme{3C分析}{3c}

\begin{stumble}
3Cの「C」が3つとも何だったか忘れやすい。
\textbf{Customer（顧客）・Competitor（競合）・Company（自社）}。
「お客さん → ライバル → うちの会社」の順。
\end{stumble}

\begin{center}
\begin{tabularx}{\textwidth}{lXX}
  \toprule
  \textbf{分析手法} & \textbf{分析の軸} & \textbf{主な用途} \\
  \midrule
  SWOT分析 & 内部/外部 × プラス/マイナス & 戦略の方向性を決める \\
  3C分析 & 顧客・競合・自社 & 市場環境を整理する \\
  4P分析 & 製品・価格・流通・販促 & マーケティング施策を決める \\
  \bottomrule
\end{tabularx}
\end{center}

\pitfallmemo{3C＝「分析」、4P＝「施策」。3Cで市場を理解して、4Pで打ち手を決める}

\begin{mondai}{3C分析における3つの「C」の組合せとして正しいものはどれか。}{basic}
  \choice{Cost, Customer, Company}
  \choice{Customer, Competitor, Company}
  \choice{Customer, Competitor, Community}
  \choice{Cost, Competitor, Communication}
\end{mondai}

\begin{mondai}{3C分析を行う主な目的として最も適切なものはどれか。}{standard}
  \choice{製品の価格を決定する}
  \choice{自社を取り巻く市場環境を整理する}
  \choice{プロジェクトの進捗を管理する}
  \choice{情報セキュリティのリスクを評価する}
\end{mondai}

\kaitoupage

\answer{B}
\begin{kaisetsu}
3C分析は\textbf{Customer（顧客）・Competitor（競合）・Company（自社）}の3つ。Costは3Cの要素ではない。
\end{kaisetsu}

\answer{B}
\begin{kaisetsu}
3C分析は\textbf{市場環境の整理}が目的。アは価格戦略（4PのPrice）、ウはプロジェクトマネジメント、エはリスクアセスメント。
\end{kaisetsu}

\hitokoto{3C＝お客さん・ライバル・自分。この順番で市場を整理。}
\qrunit{2}{2}

% ====================================================
%  2.3 PPM
% ====================================================
\mondaipage
\unittheme{PPM}{ppm}

\begin{stumble}
PPMの4象限の名前は覚えやすいが、\textbf{軸が何か}を忘れがち。
横軸＝市場占有率、縦軸＝市場成長率。
\end{stumble}

\begin{center}
\begin{tabularx}{\textwidth}{lXX}
  \toprule
  & \textbf{占有率 高い} & \textbf{占有率 低い} \\
  \midrule
  \textbf{成長率 高い} & 花形（Star） & 問題児（Question Mark） \\
  \textbf{成長率 低い} & 金のなる木（Cash Cow） & 負け犬（Dog） \\
  \bottomrule
\end{tabularx}
\end{center}

\pitfallmemo{「花形」は儲かっているように見えるが投資も必要。利益が最も安定するのは「金のなる木」}

\begin{mondai}{PPMの2つの軸として正しい組合せはどれか。}{basic}
  \choice{売上高と利益率}
  \choice{市場成長率と市場占有率}
  \choice{顧客数と製品数}
  \choice{技術力と価格競争力}
\end{mondai}

\begin{mondai}{PPMにおいて、市場成長率が低く市場占有率が高い事業を何と呼ぶか。}{standard}
  \choice{花形}
  \choice{問題児}
  \choice{金のなる木}
  \choice{負け犬}
\end{mondai}

\kaitoupage

\answer{B}
\begin{kaisetsu}
PPMは\textbf{市場成長率}（縦軸）と\textbf{市場占有率}（横軸）の2軸でマトリクスを作る。
\end{kaisetsu}

\answer{C}
\begin{kaisetsu}
成長率低＝投資が少ない。占有率高＝安定収益。よって「金のなる木」。
\end{kaisetsu}
\begin{goutoubun}
\begin{itemize}
  \item ア「花形」→ 成長率高×占有率高。投資も利益も大きい
  \item イ「問題児」→ 成長率高×占有率低。将来性はあるが不確実
  \item ウ「金のなる木」→ 成長率低×占有率高。\textbf{正解}
  \item エ「負け犬」→ 成長率低×占有率低。撤退候補
\end{itemize}
\end{goutoubun}

\hitokoto{PPM＝成長率と占有率で事業を4分類。}
\qrunit{2}{3}

% ====================================================
%  2.4 バリューチェーン
% ====================================================
\mondaipage
\unittheme{バリューチェーン}{valuechain}

\begin{stumble}
主活動と支援活動の区別がつきにくい。
\textbf{主活動＝モノが動く流れ}（仕入→製造→出荷→販売→サービス）。
\textbf{支援活動＝裏方}（人事、技術、調達、管理）。
\end{stumble}

\pitfallmemo{「調達」は支援活動。仕入れる行為そのものは「購買物流」（主活動）。混同注意}

\begin{mondai}{バリューチェーン分析の目的として最も適切なものはどれか。}{basic}
  \choice{顧客・競合・自社の3視点で市場を分析する}
  \choice{事業活動のどこで付加価値が生まれているかを明らかにする}
  \choice{市場成長率と市場占有率で事業を分類する}
  \choice{内部環境と外部環境を整理する}
\end{mondai}

\begin{mondai}{バリューチェーンの「支援活動」に該当するものはどれか。}{standard}
  \choice{製造}
  \choice{出荷物流}
  \choice{人的資源管理}
  \choice{販売・マーケティング}
\end{mondai}

\kaitoupage

\answer{B}
\begin{kaisetsu}
バリューチェーンは\textbf{付加価値の源泉を特定する}フレームワーク。アは3C分析、ウはPPM、エはSWOT分析。
\end{kaisetsu}

\answer{C}
\begin{goutoubun}
\begin{itemize}
  \item ア「製造」→ 主活動
  \item イ「出荷物流」→ 主活動
  \item ウ「人的資源管理」→ 支援活動。\textbf{正解}
  \item エ「販売・マーケティング」→ 主活動
\end{itemize}
\end{goutoubun}

\hitokoto{バリューチェーン＝モノの流れ（主活動）と裏方（支援活動）に分けて付加価値を可視化。}
\qrunit{2}{4}

\section{マーケティング}

\mondaipage
\unittheme{マーケティングミックス（4P）}{4p}

\begin{keypoint}
\begin{tabularx}{\textwidth}{lXl}
  \toprule
  \textbf{4P} & \textbf{内容} & \textbf{例} \\
  \midrule
  Product（製品） & 何を売るか & 機能、品質、デザイン \\
  Price（価格） & いくらで売るか & 定価、割引、サブスク \\
  Place（流通） & どこで売るか & 店舗、EC、代理店 \\
  Promotion（販促） & どう知らせるか & 広告、SNS、展示会 \\
  \bottomrule
\end{tabularx}
\end{keypoint}

\pitfallmemo{Placeを「店舗の場所」と直訳しがち → 正しくは「流通チャネル」。ネット通販もPlace}

\begin{mondai}{マーケティングミックスの4Pに含まれないものはどれか。}{basic}
  \choice{Product}
  \choice{Price}
  \choice{Process}
  \choice{Promotion}
\end{mondai}

\begin{mondai}{4Pの「Place」の説明として最も適切なものはどれか。}{trick}
  \choice{店舗の所在地のこと}
  \choice{製品を顧客に届けるための流通経路のこと}
  \choice{広告を掲載する媒体のこと}
  \choice{製品を製造する工場の立地のこと}
\end{mondai}

\kaitoupage

\answer{C}
\begin{kaisetsu}
4Pは\textbf{Product・Price・Place・Promotion}。Processは含まれない。
\end{kaisetsu}

\answer{B}
\begin{kaisetsu}
Place＝流通チャネル全体を指す。ネット通販もPlaceに含まれる。「場所」と直訳するとアを選んでしまうひっかけ。
\end{kaisetsu}

\hitokoto{4P＝製品・価格・流通・販促。「何を、いくらで、どこで、どう売るか」}
\qrunit{2}{5}

\section{財務と会計}

\mondaipage
\unittheme{財務三表（PL・BS・CF）}{financial}

\begin{center}
\begin{tabularx}{\textwidth}{lXXX}
  \toprule
  & \textbf{PL（損益計算書）} & \textbf{BS（貸借対照表）} & \textbf{CF（キャッシュフロー）} \\
  \midrule
  何を見る？ & 儲け & 財産の状態 & 現金の動き \\
  期間/時点 & 一定期間 & ある時点 & 一定期間 \\
  たとえ & 通知表 & 健康診断 & 家計簿 \\
  \bottomrule
\end{tabularx}
\end{center}

\pitfallmemo{BSの左側（資産）＝右側（負債＋純資産）。バランスするからBalance Sheet}

\begin{mondai}{損益計算書（PL）が表すものとして最も適切なものはどれか。}{basic}
  \choice{ある時点の資産・負債・資本の残高}
  \choice{一定期間における収益と費用、そして利益}
  \choice{一定期間における現金の増減}
  \choice{事業ポートフォリオの位置づけ}
\end{mondai}

\begin{mondai}{貸借対照表（BS）の右側に記載されるものの組合せとして正しいものはどれか。}{standard}
  \choice{資産と負債}
  \choice{負債と純資産}
  \choice{収益と費用}
  \choice{資産と純資産}
\end{mondai}

\kaitoupage

\answer{B}
\begin{kaisetsu}
PLは\textbf{一定期間の収益・費用・利益}を表す。アはBS、ウはCF、エはPPM。
\end{kaisetsu}

\answer{B}
\begin{goutoubun}
\begin{itemize}
  \item ア → 資産は左側（借方）。右側ではない
  \item イ → 右側（貸方）＝負債＋純資産。\textbf{正解}
  \item ウ → 収益と費用はPLの項目
  \item エ → 資産は左側
\end{itemize}
\end{goutoubun}

\hitokoto{PL＝期間の儲け、BS＝時点の財産、CF＝期間の現金の動き。}
\qrunit{2}{6}

\section{法務}

\mondaipage
\unittheme{IT関連の契約形態}{contract}

\begin{center}
\begin{tabularx}{\textwidth}{lXXX}
  \toprule
  & \textbf{請負契約} & \textbf{準委任契約} & \textbf{派遣契約} \\
  \midrule
  完成責任 & \textbf{あり} & なし & なし \\
  指揮命令 & 受注側 & 受注側 & \textbf{発注側} \\
  義務 & 成果物の納品 & 善管注意義務 & 労働力の提供 \\
  報酬の根拠 & 成果物 & 作業時間/工数 & 労働時間 \\
  \bottomrule
\end{tabularx}
\end{center}

\pitfallmemo{派遣は「指揮命令が発注側」。請負・準委任で発注側が直接指示したら偽装請負（違法）}

\begin{mondai}{準委任契約と請負契約の違いとして最も適切なものはどれか。}{basic}
  \choice{準委任は完成責任を負い、請負は善管注意義務のみを負う}
  \choice{請負は完成責任を負い、準委任は善管注意義務のみを負う}
  \choice{両者とも完成責任を負う}
  \choice{両者とも善管注意義務のみを負う}
\end{mondai}

\begin{mondai}{請負契約で開発を受注した場合、開発作業中の指揮命令権があるのは誰か。}{standard}
  \choice{発注者（顧客企業）}
  \choice{受注者（開発企業）}
  \choice{PMO}
  \choice{発注者と受注者の共同}
\end{mondai}

\begin{mondai}{発注側の企業が、受注側から来ているエンジニアに対して直接業務指示を行っている。この契約形態として適切なものはどれか。}{trick}
  \choice{請負契約}
  \choice{準委任契約}
  \choice{労働者派遣契約}
  \choice{売買契約}
\end{mondai}

\kaitoupage

\answer{B}
\begin{kaisetsu}
\textbf{請負＝完成責任あり}。\textbf{準委任＝善管注意義務のみ}。結果を約束するか、プロセスを約束するか。
\end{kaisetsu}

\answer{B}
\begin{goutoubun}
\begin{itemize}
  \item ア → 発注者に指揮命令権があるのは\textbf{派遣契約}
  \item イ → 請負では\textbf{受注側}が自律的に作業。\textbf{正解}
  \item ウ → PMOは支援組織であり契約上の指揮命令権とは別
  \item エ → 共同指揮は契約形態としてない
\end{itemize}
\end{goutoubun}

\answer{C}
\begin{kaisetsu}
「発注側が直接指示」＝\textbf{派遣契約}でなければ違法。請負・準委任で発注側が直接指示をしたら偽装請負。試験では「誰が指示しているか」で契約形態を判別させる問題が頻出。
\end{kaisetsu}

\hitokoto{請負＝結果を約束、準委任＝プロセスを約束、派遣＝発注側が指揮。}
\qrunit{2}{7}

%  2.8 知的財産権
% ====================================================
\mondaipage
\unittheme{知的財産権}{ip}

\begin{center}
\begin{tabularx}{\textwidth}{lXXX}
  \toprule
  & \textbf{著作権} & \textbf{特許権} & \textbf{商標権} \\
  \midrule
  保護対象 & 表現（文章・音楽・プログラム） & 発明（技術的アイデア） & マーク・ロゴ \\
  登録 & 不要（自動発生） & 必要（出願・審査） & 必要（出願・審査） \\
  保護期間 & 死後70年 & 出願から20年 & 登録から10年（更新可） \\
  \bottomrule
\end{tabularx}
\end{center}

\pitfallmemo{プログラムは著作権で保護される。アルゴリズム（考え方）は著作権では保護されない}

\begin{mondai}{著作権の説明として最も適切なものはどれか。}{basic}
  \choice{登録しなければ権利が発生しない}
  \choice{作品を創作した時点で自動的に権利が発生する}
  \choice{保護期間は出願日から20年である}
  \choice{技術的なアイデアを保護する権利である}
\end{mondai}

\begin{mondai}{プログラムの著作権として正しいものはどれか。}{trick}
  \choice{プログラム言語そのものは著作権で保護される}
  \choice{プログラムのソースコードは著作権で保護される}
  \choice{アルゴリズムは著作権で保護される}
  \choice{データベースの構造は著作権で保護されない}
\end{mondai}

\kaitoupage

\answer{B}
\begin{kaisetsu}
著作権は\textbf{無方式主義}。創作した瞬間に自動で権利が発生。アは特許権等、ウは特許権の保護期間、エは特許権の説明。
\end{kaisetsu}

\answer{B}
\begin{goutoubun}
\begin{itemize}
  \item ア → プログラム言語そのもの → 保護\textbf{されない}
  \item イ → ソースコード → 保護\textbf{される}。\textbf{正解}
  \item ウ → アルゴリズム → 著作権では保護\textbf{されない}
  \item エ → 選択・配列に創作性があれば保護される
\end{itemize}
\end{goutoubun}

\hitokoto{著作権＝登録不要・表現を保護。特許権＝登録必要・発明を保護。}

\section*{章末ミニ模試}

\begin{minimoshi}{ストラテジ系}

以下の5問を制限時間\textbf{8分}で解いてください。

\begin{mondai}{企業の内部環境と外部環境を分析するフレームワークはどれか。}{standard}
  \choice{3C分析}

\subsection*{解答}

\begin{enumerate}
  \item \textbf{ウ}（SWOT分析）— 内部と外部の両面を分析するのはSWOT。
  \item \textbf{ウ}（金のなる木）— 成長率低×占有率高。投資が少なく利益が安定。
  \item \textbf{ウ}（出願・審査で登録）— 特許権は登録が必要。アは著作権、イは著作権の保護期間、エは商標権。
  \item \textbf{イ}（成果物の完成責任）— 請負＝完成責任あり。アは準委任、ウ・エは派遣。
  \item \textbf{ウ}（Place）— Place＝流通チャネル。「場所」と直訳しないこと。
\end{enumerate}
