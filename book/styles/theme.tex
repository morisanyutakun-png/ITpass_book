% =============================================================
% theme.tex
% デザインシステムの「トークン」定義
% 色・TikZ 既定・装飾定数を一元管理する。
% ここを書き換えるだけで誌面全体の印象が変えられること。
% =============================================================

% ---- 基本パレット ----
\definecolor{themeMain}{HTML}{1F3A5F}   % 濃紺：メイン（章・強調）
\definecolor{themeSub}{HTML}{3C5A80}    % 青灰：サブ（subsection など）
\definecolor{themeAccent}{HTML}{0E7C86} % ティール：リンク・要点
\definecolor{themeMuted}{HTML}{7A8698}  % 中間グレー：副次情報
\definecolor{themeInk}{HTML}{1C2430}    % 本文テキスト
\definecolor{themeRule}{HTML}{C8D0DA}   % 細罫線
\definecolor{themeHair}{HTML}{E4E9EF}   % 極細罫線
\definecolor{themePage}{HTML}{FFFFFF}   % 紙面

% ---- 後方互換（旧色名）----
\definecolor{themeAlert}{HTML}{C8662A}  % 旧：alert（= pitBar と同系）
\definecolor{themeInfoBg}{HTML}{F7F9FC} % 旧：info bg
\definecolor{themeKeyBg}{HTML}{EAF5F4}
\definecolor{themePitBg}{HTML}{FBEFE3}
\definecolor{themeCmpBg}{HTML}{ECF1F7}
\definecolor{themeSumBg}{HTML}{FFF6DC}

% ---- 役者カラー（ボックス毎）----
% 要点：ティール
\definecolor{pointBar}{HTML}{0E7C86}
\definecolor{pointBG}{HTML}{EAF5F4}
% つまずき：暖オレンジ（赤を避け参考書らしく）
\definecolor{pitBar}{HTML}{C8662A}
\definecolor{pitBG}{HTML}{FBEFE3}
% 本試験出題：紫紺
\definecolor{examBar}{HTML}{5B4A8B}
\definecolor{examBG}{HTML}{F1ECF5}
% 学習アドバイス：グレー
\definecolor{adviceBar}{HTML}{6B7684}
\definecolor{adviceBG}{HTML}{F3F5F7}
% 比較：青灰
\definecolor{cmpBar}{HTML}{3C5A80}
\definecolor{cmpBG}{HTML}{ECF1F7}
% まとめ：紺＋クリーム
\definecolor{sumBar}{HTML}{1F3A5F}
\definecolor{sumBG}{HTML}{FFF6DC}

% ---- 章扉用 ----

% ---- 装飾定数（寸法トークン）----

% ---- TikZ の既定 ----
\tikzset{
  node distance=10mm and 12mm,
  every node/.style={font=\small},
  boxnode/.style={
    draw=themeMain, thick, fill=white, rounded corners=2pt,
    inner sep=5pt, align=center, minimum height=8mm
  },
  keynode/.style={
    draw=themeAccent, thick, fill=pointBG, rounded corners=2pt,
    inner sep=5pt, align=center, minimum height=8mm
  },
  subnode/.style={
    draw=themeRule, thin, fill=white, rounded corners=2pt,
    inner sep=4pt, align=center, font=\footnotesize
  },
  groupbg/.style={
    draw=themeHair, fill=cmpBG, rounded corners=3pt, inner sep=6pt
  },
  arr/.style={-{Latex[length=2mm]}, themeSub, thick},
  arr2/.style={{Latex[length=2mm]}-{Latex[length=2mm]}, themeSub, thick},
}
