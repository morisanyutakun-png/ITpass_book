% --- カタカナラベル ---
\newcommand{\katalabel}[1]{%
  \ifcase#1\or ア\or イ\or ウ\or エ\or オ\or カ\fi
}

% --- 4択問題環境（試験問題風） ---
\newenvironment{mondai}[2]{%
  \refstepcounter{questionctr}%
  \par\vspace{8pt}%
  \noindent
  \textbf{\sffamily 問\thequestionctr}%
  \hfill
  \ifstrequal{#2}{basic}{\levelbasic}{}%
  \ifstrequal{#2}{standard}{\levelstandard}{}%
  \ifstrequal{#2}{trick}{\leveltrick}{}%
  \par\vspace{2pt}%
  \noindent #1%
  \par\vspace{4pt}%
  \setcounter{choicectr}{0}%
}{%
  \par\vspace{6pt}%
  {\color{RuleGray}\rule{\textwidth}{0.15pt}}%
}

% --- 選択肢（ア/イ/ウ/エ） ---
\newcommand{\choice}[1]{%
  \stepcounter{choicectr}%
  \par\noindent\hspace{1.5em}%
  \textbf{\katalabel{\value{choicectr}}}\hspace{0.6em}#1%
}
