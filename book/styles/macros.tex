% ---- サポートサイト URL（QR 連携の一元管理） ----
% GitHub Pages の公開URLが決まったら、ここだけを書き換える。
% \detokenize を \edef で強制適用し、URL 中の "_" 等が数式扱いされないようにする。
\edef\supportbaseurl{\detokenize{https://morisanyutakun-png.github.io/ITpass_book/}}

% 章補助URL: \supporturl{errata} → <base>errata.html
\NewDocumentCommand{\supporturl}{m}{\supportbaseurl\detokenize{#1}.html}

% ---- QR コード ----
% \supportqr{quiz}{確認テスト} のように使う
% qrcode は引数を展開しないため、\edef で事前に URL 文字列を確定させる。
\makeatletter
\newcommand{\itp@qrurl}{}
\NewDocumentCommand{\supportqr}{m m}{%
  \begingroup
  \edef\itp@qrurl{\supportbaseurl\detokenize{#1}.html}%
  \begin{center}
    \begin{tcolorbox}[
      enhanced, colback=white, colframe=themeRule,
      boxrule=0.4pt, arc=2pt, left=6pt,right=6pt,top=6pt,bottom=6pt,
      width=0.78\linewidth
    ]
      \begin{minipage}{0.30\linewidth}
        \centering
        \qrcode[height=22mm]{\itp@qrurl}
      \end{minipage}%
      \hfill
      \begin{minipage}{0.62\linewidth}
        \small
        \textbf{#2}\\[2pt]
        スマートフォンのカメラで読み取ると、\\
        サポートサイトの該当ページへ移動します。\\[2pt]
        {\tiny\urlstyle{tt}\expandafter\nolinkurl\expandafter{\itp@qrurl}}
      \end{minipage}
    \end{tcolorbox}
  \end{center}%
  \endgroup
}
\makeatother

% ---- 章末確認（4 択） ----
% \checkquestion{問題文}{選択肢A}{選択肢B}{選択肢C}{選択肢D}{正解ラベル}{解説}
\NewDocumentCommand{\checkquestion}{m m m m m m m}{%
  \begingroup
  \par\medskip
  \noindent\textbf{確認問題}\quad #1\par
  \begin{enumerate}[label=\textbf{\Alph*.},leftmargin=2em,itemsep=1pt,topsep=2pt]
    \item #2
    \item #3
    \item #4
    \item #5
  \end{enumerate}
  \begin{adviceinline}
    \textbf{正解}\ #6\par
    \textbf{解説}\ #7
  \end{adviceinline}
  \par\medskip
  \endgroup
}
