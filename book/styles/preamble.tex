% =============================================================
% 章扉：\@makechapterhead を再定義
% 「CHIP＋THICKBAR＋TITLE＋ACCENTRULE」でメリハリを出す。
% =============================================================
\makeatletter
\renewcommand{\@makechapterhead}[1]{%
  \vspace*{10pt}%
  \noindent
  \begin{minipage}[t]{\linewidth}
    % 上段: 章番号チップ
    \tikz[baseline=(n.base)]{
      \node[
        fill=chapChip, text=chapChipText,
        inner xsep=10pt, inner ysep=5pt,
        font=\sffamily\small\bfseries, rounded corners=1pt
      ] (n) {\ CHAPTER\ \ \@arabic\c@chapter\ };
    }%
    \par
    \vspace{8pt}%
    % 太バー（章の"存在感"）
    {\color{chapChip}\rule{0.38\linewidth}{5pt}}%
    {\color{chapTagLine}\rule{0.22\linewidth}{5pt}}%
    \par
    \vspace{14pt}%
    % メインタイトル
    {\raggedright\LARGE\bfseries\color{themeMain} #1\par}%
    \vspace{8pt}%
    {\color{themeHair}\rule{\linewidth}{0.4pt}}%
  \end{minipage}
  \par
  \vskip 22\p@
}
\makeatother

% =============================================================
% 見出しデザイン（section / subsection のみ titlesec 適用）
% =============================================================
% section：色付きナンバーチップ＋本文色
\titleformat{\section}
  {\normalfont\Large\bfseries}
  {%
    \tikz[baseline=(s.base)]{
      \node[
        fill=themeAccent, text=white,
        inner xsep=7pt, inner ysep=2.5pt,
        font=\sffamily\normalsize\bfseries, rounded corners=1pt
      ] (s) {\thesection};
    }%
  }
  {0.6em}
  {\color{themeMain}}

% subsection：左バー＋ミュート色
\titleformat{\subsection}
  {\normalfont\large\bfseries}
  {%
    {\color{themeAccent}\rule[-0.1ex]{\actorBarWidth}{1.0em}}%
    \hspace{0.45em}%
    {\color{themeSub}\thesubsection}%
  }
  {0.45em}
  {\color{themeMain}}

\titlespacing*{\section}{0pt}{20pt plus 2pt}{8pt}
\titlespacing*{\subsection}{0pt}{14pt plus 2pt}{5pt}

% section 見出しの直後に章の"呼吸"を入れるための小設定
\renewcommand{\baselinestretch}{1.06}

% =============================================================
% 目次デザイン：章は大きく・節は小さく・ドットリーダーは極細
% =============================================================
\setcounter{tocdepth}{1}  % subsection を目次に出さない

% 目次タイトルを柔らかく
\renewcommand{\contentsname}{目次}

% 章エントリ
\titlecontents{chapter}[0pt]
  {\addvspace{12pt}}
  {%
    \large\bfseries\color{themeMain}
    \makebox[2.4em][l]{\thecontentslabel}%
  }
  {\large\bfseries\color{themeMain}}
  {\normalfont\color{themeMuted}\ \titlerule*[0.7pc]{\footnotesize.}\ \contentspage}
  [\vspace{1pt}{\color{themeHair}\rule{\linewidth}{0.3pt}}]

% 節エントリ
\titlecontents{section}[2em]
  {\addvspace{2pt}\small}
  {\color{themeSub}\makebox[2.2em][l]{\thecontentslabel}}
  {}
  {\normalfont\footnotesize\color{themeMuted}\ \titlerule*[0.7pc]{\tiny.}\ \contentspage}
